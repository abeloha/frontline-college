module.exports = {
  apps : [{
    name: "frontline",

    args: "start",
    env: {
      PORT: 3010
    }
  }]

sudo nano /etc/nginx/sites-available/gcmsapis.com
server {
    listen 80;
    server_name frontlinecollege.com.ng www.frontlinecollege.com.ng;

    location / {
        proxy_pass http://127.0.0.1:3010;
        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
sudo ln -s /etc/nginx/sites-available/gcmsapis.com /etc/nginx/sites-enabled/

sudo nginx -t
sudo systemctl reload nginx
