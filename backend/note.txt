step 1: on the server, create a systemd unit file to keep your application running in the background

nano /lib/systemd/system/frontline_backend.service

[Unit]
Description=frontline_backend

[Service]
Type=simple
Restart=always
RestartSec=5s
ExecStart=/var/www/frontline-college/backend/main
WorkingDirectory=/var/www/frontline-college/backend

[Install]
WantedBy=multi-user.target


Save and run:
 service frontline_backend start
 service frontline_backend status

 Step 2 — Setting Up a Reverse Proxy with Nginx
Note: This is only neccessary to use ssl and to connect domain. Otherwise, if the firewall allows the port, you can access the app via the ip


sudo nano /etc/nginx/sites-available/api.frontlinecollege.com.ng

 server {
    server_name api.frontlinecollege.com.ng;

     location / {

        
        proxy_pass http://localhost:3011;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

    }
}



link
sudo ln -s /etc/nginx/sites-available/api.frontlinecollege.com.ng /etc/nginx/sites-enabled/api.frontlinecollege.com.ng

sudo nginx -t
sudo systemctl restart nginx
